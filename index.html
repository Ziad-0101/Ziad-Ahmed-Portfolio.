A repository abstraction in Core and a query handler that depends only on it. A pattern sketch, not code from a specific project.

// Core: abstraction the application layer depends on
public interface IRepository<T> where T : class
{
    Task<T?> GetByIdAsync(int id, CancellationToken ct);
    Task AddAsync(T entity, CancellationToken ct);
}

// Application: CQRS read side
public record GetListingQuery(int Id) : IRequest<ListingDto?>;

public class GetListingHandler(IRepository<Listing> repo)
    : IRequestHandler<GetListingQuery, ListingDto?>
{
    public async Task<ListingDto?> Handle(GetListingQuery q, CancellationToken ct)
    {
        var listing = await repo.GetByIdAsync(q.Id, ct);
        return listing is null ? null : new ListingDto(listing.Id, listing.Title, listing.Rent);
    }
}

Issuing a signed, short-lived access token with identity and role claims. A pattern sketch.

public string CreateToken(User user)
{
    var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_options.Secret));
    var claims = new[]
    {
        new Claim(JwtRegisteredClaimNames.Sub, user.Id.ToString()),
        new Claim(ClaimTypes.Role, user.Role)
    };

    var token = new JwtSecurityToken(
        issuer: _options.Issuer,
        audience: _options.Audience,
        claims: claims,
        expires: DateTime.UtcNow.AddMinutes(30),
        signingCredentials: new SigningCredentials(key, SecurityAlgorithms.HmacSha256));

    return new JwtSecurityTokenHandler().WriteToken(token);
}
